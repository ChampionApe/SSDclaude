\subsection{The politico-economic equilibrium with log}\label{\refsec:PEELOG}
Let $z_t$ denote the marginal effect of the bounded policy on the political objective,
\begin{align}\label{\refeq:LOG:fast}
    z_t &= \omega\sum_{j}\left(\gamma_{t-1,j}p_{t-1,j}\mu_{t-1,j} \frac{d\upsilon_{2,t}^j}{d\tau_t}\right)+\nu_t\sum_j\left(\gamma_{t,j}\mu_{t,j} \frac{d\upsilon_{1,t}^j}{d\tau_t}\right),
\end{align}
with the marginal effects on indirect utilities from \eqref{\refmodeleq:PEELOG} for $t<T$ and \eqref{\refmodeleq:terminalPEELOG} for $t=T$.

\smalltitle{The problem carries no endogenous state, and the system is triangular.}
The savings shares $s_{t-1,i}/s_{t-1}$ that the old carry into period $t$ are closed-form functions of $\tau_t$, at the previous period's parameters (\cref{\refmodelsec:PEELOG}), and no other predetermined quantity enters $z_t$. Of the terms of $z_t$, the young's depend on $\tau_t$ alone and the old formal one on $\tau_t$ directly and through the shares; only the old informal term reaches into the future, through the level of $\Theta_{h,t}$, which depends on $\tau_{t+1}$ \eqref{\refeq:auxiliary:Thetah}. No term depends on a lag of $\tau$. Hence
\begin{align}\label{\refeq:recursive}
    z_t = z_t\left(\tau_t, \tau_{t+1}\right), \qquad\text{and}\qquad z_T = z_T(\tau_T),
\end{align}
the latter because the terminal period uses \eqref{\refeq:auxiliary:ThetahT}. The system is triangular: a \emph{backward recursion} solves it exactly, each period a \emph{scalar} problem in $\tau_t\in[l,u]$ given the solved $\tau_{t+1}$, with robust one-dimensional methods and no initial guess for the path. The property relies on $\theta_t$ and $\epsilon_t$ being exogenous.

\smalltitle{Boundaries.}
Both $d\upsilon_{1,t}^j/d\tau_t$ and $\partial \ln(\Theta_{h,t})/\partial\tau_t$ carry a factor $1/(1-\tau_t)$, so $z_t\rightarrow -\infty$ as $\tau_t\rightarrow 1$ and the grid stops at $u<1$; the default grid has 101 nodes on $[10^{-4}, 1-10^{-4}]$. The upper corner is then not chosen in practice, and what remains is to detect a lower corner and to choose between several interior crossings, which the selection of \cref{\refsec:RobustRoot} does.

\begin{alg}[Gradient-based roots for the politico-economic equilibrium with log preferences]
\label{\refalg:LOG:fast}
Let $\tilde{\bm{\tau}} \in \mathbb{R}^{T}$ be a candidate vector of policies and $\bm{\tau}$ its bounded version, each $\tau_t \in [l, u]$. Stack \eqref{\refeq:root} across periods, with $z_t$ from \eqref{\refeq:LOG:fast}, and solve for the $\tilde{\bm{\tau}}$ that zeroes the stacked system with a standard root finder, started from the previous solution or a constant path at the baseline tax; if the residual misses the tolerance, restart once from the point reached. Report the bounded $\bm{\tau}$. The algorithm has no fallback and presumes that the first order condition is well behaved: its value is speed and precision when it works, not robustness.
\end{alg}

\begin{alg}[Backward grid search]\label{\refalg:LOG:gridsearch}
Fix $\bm{\theta},\bm{\epsilon}$ and a grid $\mathcal{T}$ on $[l,u]$. Exploiting \eqref{\refeq:recursive}, solve backwards.

\smallskip\noindent\emph{Terminal period.} Evaluate $z_T$ on all of $\mathcal{T}$, construct the candidate set $\mathcal{C}_T$ from \eqref{\refeq:candidates}, apply the equilibrium test \eqref{\refeq:equilibriumTest} at each candidate's own savings shares, and select among the equilibria as in \cref{\refsec:RobustRoot}.

\smallskip\noindent\emph{Recursion, $t = T-1,...,1$.} Given the solved $\tau_{t+1}$:
\begin{enumerate}
    \item \emph{Window.} Let $\mathcal{T}_t\subseteq\mathcal{T}$ consist of the nodes from $\Delta^-$ below to $\Delta^+$ above the node of $\tau_{t+1}$. The window need not be symmetric: where the policy path is known to be monotone, $\Delta^-\neq\Delta^+$ saves evaluations.
    \item \emph{Evaluate and select.} Evaluate $z_t(\cdot,\tau_{t+1})$ on $\mathcal{T}_t$ and apply the terminal-period rule.
    \item \emph{Expand.} If the selected $\tau_t$ falls on an endpoint of $\mathcal{T}_t$ that is not an endpoint of $\mathcal{T}$, the window may have excluded the maximum: double the window on that side and return to step 2.
\end{enumerate}
The cost is $O(T\cdot M)$ evaluations of closed-form expressions, with no derivatives and no initial guess, and the located interior roots are as accurate as the grid.
\end{alg}

\smalltitle{The default combines the two.}
The default solver runs \cref{\refalg:LOG:fast} and then \cref{\refalg:LOG:gridsearch} on the full grid, which applies the selection rule and reports the candidate and equilibrium counts. Where the gradient solution lies within one grid cell of the selected equilibrium, it is kept. Otherwise, or where the gradient solve failed, the grid solution is polished by \cref{\refalg:LOG:fast} started from it, and where the polish fails the grid solution is returned, flagged as grid-accurate. The grid search locates the correct branch of a problem with corners and possible multiplicity, and the gradient step attains the solver's tolerance on a problem that, near the located solution, has neither.
