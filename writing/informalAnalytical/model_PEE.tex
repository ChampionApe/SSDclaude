\subsection{The politico-economic equilibrium with LOG preferences}\label{\refsec:PEELOG}
Let $\upsilon_{1,t}^j$ and $\upsilon_{2,t}^j$ denote the indirect utilities of young and old households of type $j$. Policy is set without commitment: each period an election aggregates preferences by probabilistic voting, which maximizes a weighted sum of indirect utilities. Let $\tau^{t+1}(z_t)$ denote a continuation policy, with $z_t$ a vector of states. The political problem is
\begin{align*}
    &\max_{\tau_t}\mbox{ }\mathcal{W}_t(z_{t-1}) \equiv \omega\sum_j\left(\gamma_{t-1,j} p_{t-1,j} \mu_{t-1,j} \upsilon_{2,t}^j\right)+\nu_t\sum_j\left(\gamma_{t,j}\mu_{t,j} \upsilon_{1,t}^j\right) \\
    &\text{s.t. competitive equilibrium and }\tau_{t+1} = \tau^{t+1}(z_t),
\end{align*}
where $\tau^{t+1}(z_t)$ is identified by the same problem at $t+1$, $\omega$ is the political weight of the old, and $\mu_{t,j}$ the propensity to vote of type $j$. The old at $t$ vote with the propensities $\mu_{t-1,j}$ they had when young: propensities are attached to a generation, not to a period.

With $\rho = 1$, the discount function simplifies to $B_{t+1}^i = \beta_{t,i}$ and the politico-economic equilibrium has a closed form without continuation policy functions. Every consumption function is log-linear in $s_{t-1}/\nu_t$ with an exponent that policy does not move \eqref{\refeq:EE:sigma_ci}, so the state enters each indirect utility additively and drops out of the first order condition, and the savings shares $s_{t-1,i}/s_{t-1}$ that remain in it are, in equilibrium, closed-form functions of $\tau_t$ \eqref{\refeq:EE:si_s}. The political problem therefore carries no endogenous state, $z_t = \emptyset$. In general, the first order condition for $\tau_t$ is
\begin{align*}
    \omega\sum_{j}\left(\gamma_{t-1,j}p_{t-1,j}\mu_{t-1,j} \frac{d\upsilon_{2,t}^j}{d\tau_t}\right)+
    \nu_t\sum_j\left(\gamma_{t,j}\mu_{t,j} \frac{d\upsilon_{1,t}^j}{d\tau_t}\right) = 0,
\end{align*}
where the total derivatives include the effects of policy through the economic equilibrium and those through the continuation policy.

\begin{subequations}\label{\refeq:PEELOG}
In the log case there are no effects through the continuation policy, and the condition simplifies. For formal young households the Euler equation $c_{2,t+1}^i = \beta_{t,i}(R_{t+1}/p_t)\tilde{c}_{1,t}^i$ gives $\upsilon_{1,t}^i = (1+\beta_{t,i})\ln\left(\tilde{c}_{1,t}^i\right)+\beta_{t,i} \ln\left(R_{t+1}/p_t\right)+\beta_{t,i} \ln\left(\beta_{t,i}\right)$. Since $\tilde{c}_{1,t}^i$ is log-separable in $s_{t-1}$, the formula for the interest rate gives
\begin{align}
    \frac{d\upsilon_{1,t}^i}{d\tau_t} = -\frac{1}{1-\tau_t}\frac{1+\xi}{1+\alpha\xi}\left(1+\beta_{t,i}\frac{\alpha(1+\xi)}{1+\alpha\xi}\right).
\end{align}
For the informal young, the current tax works only through the future informal wage:
\begin{align}
    \frac{d\upsilon_{1,t}^0}{d\tau_t} = -\frac{\beta_{t,0}}{1-\tau_t}\frac{\alpha(1+\xi)^2}{(1+\alpha\xi)^2}.
\end{align}
For the old, the effect works through $\tilde{c}_{2,t}^j$, again log-separable in $s_{t-1}$:
\begin{align}
    \frac{d\upsilon_{2,t}^i}{d\tau_t} =& (1-\alpha)\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial \tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{p_{t-1}}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{p_{t-1}\tau_t}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)} \\
    \frac{d\upsilon_{2,t}^0}{d\tau_t} =& \frac{\dfrac{(1-\alpha)\nu_t\epsilon_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}\left(1+(1-\alpha)\tau_t\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial\tau_t}\right)}{\frac{\left(\chi_{t-1}\eta_{t-1,0}\right)^{1+\xi}}{(1+\xi)X_{t-1,0}^{\xi}}\left(\frac{1-\alpha}{\Gamma_{h,t}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\dfrac{(1-\alpha)\nu_t\epsilon_t\tau_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}},
\end{align}
where
\begin{align*}
    \frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial \tau_t} = -\frac{1}{1-\tau_t}\frac{\xi}{1+\alpha\xi}.
\end{align*}
\end{subequations}



\subsection{The politico-economic equilibrium with CRRA preferences}\label{\refsec:PEE}
With CRRA preferences the first order condition is the same, but the total derivatives of indirect utilities change. As \cref{\refnumsec} shows, we approximate these derivatives numerically and combine them with grid searches. For the young this requires only indirect utilities. The old need one step more: the electorate takes their savings shares $s_{t-1,i}/s_{t-1}$ as given, so the derivative must hold them fixed, but in equilibrium they are the closed-form function \eqref{\refeq:EE:si_s} of the tax that savers anticipated, so substituting that function after differentiating replaces a grid over the distribution of past savings with the single state $s_{t-1}$.


\begin{subequations}\label{\refeq:PEE}
For the formal young the Euler equation gives
\begin{align}
    \upsilon_{1,t}^i = (1+B_{t+1}^i)\frac{\left(\tilde{c}_{1,t}^i\right)^{1-1/\rho}}{1-1/\rho}.
\end{align}
The indirect utility of the informal young cannot be reduced further, as they consume hand-to-mouth and do not smooth consumption over time:
\begin{align}
    \upsilon_{1,t}^0 = \frac{1}{1-1/\rho}\left[ \left(\tilde{c}_{1,t}^0\right)^{1-1/\rho}+\beta_{t,0} \left(\tilde{c}_{2,t+1}^0\right)^{1-1/\rho}\right]
\end{align}
For the formal old the derivative holds the shares fixed:
\begin{align}
    \dfrac{d \upsilon_{2,t}^i}{d \tau_t} =& \left(c_{2,t}^i\right)^{1-1/\rho}\dfrac{d\ln\left(c_{2,t}^i\right)}{d\tau_t} \\
    \frac{d\ln\left(c_{2,t}^i\right)}{d\tau_t} =& (1-\alpha)\frac{d \ln\left(h_t\right)}{d \tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{p_{t-1}}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{p_{t-1}\tau_t}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)},
\end{align}
and is then evaluated at the shares that \eqref{\refeq:EE:si_s} gives for $\tau_t$, so that only the equilibrium selection of past distributions is ever visited.

The informal old carry no savings, so their indirect utility is evaluated directly:
\begin{align}
    \upsilon_{2,t}^0 = \frac{\left(\tilde{c}_{2,t}^0\right)^{1-1/\rho}}{1-1/\rho}.
\end{align}
\end{subequations}
