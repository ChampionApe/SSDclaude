\section{Politico-economic equilibrium}\label{app:PEE}\setcounter{equation}{0}

\subsection{Proof of proposition \ref{prop:LOG:PEE}}\label{app:PEE:LOG}
By appendix \ref{app:EE:LOG}, all consumption functions are log-separable in the aggregate state $s_{t-1}$, and in all the $\Theta$-coefficient functions the only \textit{predetermined} variable that may enter the political problem as an endogenous state is the distribution of savings $s_{t-1,i}/s_{t-1}$. The current tax moves the aggregate state carried into the next period, $s_t$, but by the same log-separability its level drops out of the first order condition at $t+1$. The distribution chosen at $t$, $s_{t,i}/s_t$, depends on parameters and on the tax and the design anticipated for $t+1$ (part iii of proposition \ref{prop:LOG:EE}), but not on $\tau_t$, so that the part of the law of motion \eqref{eq:model:lawOfMotion_General} that carries the distribution does not respond to the current tax,
\begin{align*}
    \dfrac{\partial \left(s_{t,i}/s_t\right)}{\partial \tau_t} = 0, \qquad i>0.
\end{align*}
The distribution is therefore not an \textit{endogenous} state: changing $\tau_t$ has no strategic effect on future policy.

The tax at $t$ maximizes \eqref{eq:LOG:PolObj} given the inherited distribution of savings and the continuation policy, and it satisfies the first order condition
\begin{align*}
    &\omega\Bigg[\gamma_0\mu_0 \overbrace{\dfrac{\partial \ln\left(\Theta_{\tilde{c}_2,t}^0\right)}{\partial \tau_t}}^{=\mathcal{E}_{2,t}^0}+\sum_{i>0}\gamma_i\mu_i \overbrace{\dfrac{\partial \ln\left(\Theta_{c_2,t}^i\right)}{\partial \tau_t}}^{= \mathcal{E}_{2,t}^i}\Bigg]
    \\+&\nu_t\Bigg\lbrace \gamma_0\mu_0 \underbrace{\left(\dfrac{\partial \ln\left(\Theta_{\tilde{c}_1,t}^0\right)}{\partial \tau_t}+\beta\dfrac{\partial \ln\left(\Theta_{\tilde{c}_2,t+1}^0\right)}{\partial \tau_t}\right)}_{= \mathcal{E}_{1,t}^0}
    \\&\qquad+\sum_{i>0}\gamma_i\mu_i \underbrace{\left(\dfrac{\partial \ln\left(\Theta_{\tilde{c}_1,t}^i\right)}{\partial \tau_t}+\beta\dfrac{\partial \ln\left(\Theta_{c_2,t+1}^i\right)}{\partial \tau_t}\right)}_{= \mathcal{E}_{1,t}^i}\Bigg\rbrace \leq 0,
\end{align*}
with equality for an interior solution, where $\mathcal{E}_{x,t}^j$ denotes the marginal effect of the tax on the indirect utility of a household of type $j$ in generation $x$ and $\mu_j$ the propensity to vote of type $j$. Using the equilibrium consumption functions of appendix \ref{app:EE:CRRA},
\begin{align}
    \mathcal{E}_{2,t}^0 & = \frac{\dfrac{(1-\alpha)\nu_t\epsilon}{1+\gamma_0\epsilon} \Theta_{h,t}^{1-\alpha}\left(1+(1-\alpha)\tau_t\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial\tau_t}\right)}{\frac{\left(\chi\eta_{0}\right)^{1+\xi}}{(1+\xi)X_{0}^{\xi}}\left(\frac{1-\alpha}{\Gamma_{h}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\dfrac{(1-\alpha)\nu_t\epsilon\tau_t}{1+\gamma_0\epsilon} \Theta_{h,t}^{1-\alpha}} \label{eq:LOG:E20}\\
    \mathcal{E}_{2,t}^i&= (1-\alpha)\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial \tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{1}{1+\gamma_0\epsilon}\left(1+\theta_t\left[\frac{\eta_{i}^{1+\xi}}{X_{i}^{\xi}}\frac{1}{\Gamma_{h}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{\tau_t}{1+\gamma_0\epsilon}\left(1+\theta_t\left[\frac{\eta_{i}^{1+\xi}}{X_{i}^{\xi}}\frac{1}{\Gamma_{h}}-1\right]\right)} \label{eq:LOG:E2i}\\
    \mathcal{E}_{1,t}^0 &= -\frac{\beta}{1-\tau_t}\frac{\alpha(1+\xi)^2}{(1+\alpha\xi)^2} \label{eq:LOG:E10}\\
    \mathcal{E}_{1,t}^i &= -\frac{1}{1-\tau_t}\frac{1+\xi}{1+\alpha\xi}\left(1+\beta\frac{\alpha(1+\xi)}{1+\alpha\xi}\right). \label{eq:LOG:E1i}
 \end{align}

The tax lowers the net income of the young and with it their savings, $\partial\ln(s_t)/\partial\tau_t = -\frac{1+\xi}{1+\alpha\xi}\frac{1}{1-\tau_t}$ (appendix \ref{app:EE:LOG}). An informal young household earns the informal wage, which does not depend on $\tau_t$, and in old age receives an endowment and a universal pension that both scale with $s_t^{\alpha(1+\xi)/(1+\alpha\xi)}$, which gives \eqref{eq:LOG:E10}. For a formal young household the Euler equation writes indirect utility as $(1+\beta)\ln(\tilde{c}_{1,t}^i)+\beta\ln(R_{t+1})+\beta\ln(\beta)$. Here $\tilde{c}_{1,t}^i$ is proportional to $s_t$ with a factor that does not depend on $\tau_t$ (appendix \ref{app:EE:CRRA}), and $R_{t+1} = \alpha\Theta_{h,t+1}^{1-\alpha}(s_t/\nu_{t+1})^{-(1-\alpha)/(1+\alpha\xi)}$ by \eqref{eq:factorPrices} and \eqref{eq:h}, with $\Theta_{h,t+1}$ set by policy at $t+1$ and $t+2$, which does not respond to $\tau_t$. The fall in savings lowers the consumption of the young and raises the return on their savings, which partly compensates them: $(1+\beta)-\beta\frac{1-\alpha}{1+\alpha\xi} = 1+\beta\frac{\alpha(1+\xi)}{1+\alpha\xi}$ gives \eqref{eq:LOG:E1i}. For retirees the tax has opposing effects in \eqref{eq:LOG:E2i}. The first term is negative, since the tax lowers hours and with them both the base of the pension and the return on savings, and the second is positive, the direct effect of the tax on the pension.

Write $y_i \equiv \frac{\eta_i^{1+\xi}}{X_i^{\xi}\Gamma_h} = \frac{\eta_ih_{t-1,i}}{h_{t-1}}$ for the labor income of type $i$ relative to the mean, with $\sum_{i>0}\gamma_iy_i = 1$ by \eqref{eq:hi_h}, $x_i \equiv s_{t-1,i}/s_{t-1}$ for its savings share, $p_i \equiv 1+\theta_t(y_i-1)$ for its pension relative to the average formal pension, and $a \equiv \frac{1-\alpha}{\alpha}\frac{1}{1+\gamma_0\epsilon}$. By appendix \ref{app:EE:CRRA} and $R_ts_{t-1} = \alpha\nu_th_t^{1-\alpha}(s_{t-1}/\nu_t)^{\alpha}$, a formal retiree of type $i$ consumes $c_{2,t}^i = R_ts_{t-1}D_i$ with $D_i \equiv x_i+a\tau_tp_i$. Since $\nu_tw_th_t = \frac{1-\alpha}{\alpha}R_ts_{t-1}$, the pension is $b_t^i = R_ts_{t-1}a\tau_tp_i$ by \eqref{eq:model:governmentBudget}, so the pension's share of consumption is $\sigma_t^i \equiv b_t^i/c_{2,t}^i = a\tau_tp_i/D_i$. For an informal retiree it is $\sigma_t^0 \equiv b_t^0/\tilde{c}_{2,t}^0$. Let
\begin{align*}
    \phi_t \equiv 1+(1-\alpha)\tau_t\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial\tau_t} = 1-\frac{(1-\alpha)\xi}{1+\alpha\xi}\frac{\tau_t}{1-\tau_t}
\end{align*}
be the elasticity of revenue with respect to the tax rate, so that $1-\phi_t$ is the elasticity of the tax base in absolute value. Then \eqref{eq:LOG:E20} and \eqref{eq:LOG:E2i} read
\begin{align}\label{eq:LOG:E2sigma}
    \mathcal{E}_{2,t}^0 = \frac{\phi_t\sigma_t^0}{\tau_t}, \qquad \mathcal{E}_{2,t}^i = \frac{\phi_t\sigma_t^i-(1-\phi_t)(1-\sigma_t^i)}{\tau_t} = \frac{\phi_t-1+\sigma_t^i}{\tau_t}:
\end{align}
a unit of tax raises every pension by $\phi_t/\tau_t$ in proportion, and with log utility a retiree's support is that proportion times the pension's share of consumption, less, for formal retirees, the fall in the return on their savings.

\smalltitle{Savings consistent with the tax.} The generation that is old at $t$ saved at $t-1$ for the tax and the design that pay its benefits, which in equilibrium are those in force at $t$. By part iii of proposition \ref{prop:LOG:EE}, one period back, its savings shares and consumption are
\begin{align}\label{eq:LOG:consistent}
    x_i(\tau_t) = y_i+\frac{a\tau_t(1-\theta_t)}{1+\beta}\left(y_i-1\right), \qquad D_i = y_i+a\tau_tq_i, \qquad q_i \equiv \theta_ty_i+(1-\theta_t)\frac{\beta+y_i}{1+\beta}.
\end{align}
Types below mean income save less than their income share, by the fraction $1/(1+\beta)$ of the present value of what the flat component of the pension transfers to them, and since $q_i > 0$, retirement consumption is positive and rises with relative income. Write $\mathcal{F}_t(\tau_t;x)$ for the left-hand side of the first order condition at savings shares $x = (x_i)_{i>0}$, and $G_t(\tau_t) \equiv \mathcal{F}_t(\tau_t;x(\tau_t))$ for the condition at the shares consistent with $\tau_t$. A tax is a politico-economic equilibrium at $t$ if and only if it maximizes the political objective at the shares consistent with it, which is the test of section \ref{sec:numerical}.

\begin{proposition}\label{prop:PEE:unique}
Assume log-GHH preferences and hand-to-mouth informal households, with $\xi(1-2\alpha) \leq 1$ if $\gamma_0 > 0$. Given the design and the continuation policy, $G_t$ is strictly decreasing on $[0,1)$, and the politico-economic equilibrium tax at $t$ is its zero, or $\tau_t = 0$ if $G_t(0) \leq 0$. It is the only candidate that passes the equilibrium test, and for every horizon $T$ the politico-economic equilibrium is unique. Without informal households $G_t$ involves neither the continuation policy nor $T$, and the tax at each date before $T$ is the same for every horizon.
\end{proposition}

\smalltitle{Proof of proposition \ref{prop:PEE:unique}.} By \eqref{eq:LOG:consistent}, at consistent shares the second term of \eqref{eq:LOG:E2i} is $ap_i/(y_i+a\tau_tq_i)$, positive, finite and strictly decreasing in $\tau_t$ on $[0,1)$. A higher anticipated tax lowers the savings of a retiree below mean income, but its pension rises by more than its savings income falls, so the pension's share of its consumption rises less than in proportion to the tax. The first term of \eqref{eq:LOG:E2i} and the young's terms \eqref{eq:LOG:E10} and \eqref{eq:LOG:E1i} are non-positive multiples of $1/(1-\tau_t)$, and \eqref{eq:LOG:E1i} is a negative multiple, strictly decreasing and unbounded below as $\tau_t \to 1$. The informal retirees' term \eqref{eq:LOG:E20} is the derivative of $\ln(e_t+b_t^0)$, where the endowment $e_t$ does not depend on $\tau_t$ and, by \eqref{eq:h}, the pension $b_t^0$ is proportional to $\tau_t(1-\tau_t)^m$ with $m \equiv (1-\alpha)\xi/(1+\alpha\xi)$. Its derivative in $\tau_t$ has the sign of $(e_t+b_t^0)\,\partial^2b_t^0/\partial\tau_t^2-(\partial b_t^0/\partial\tau_t)^2$, which is negative where $b_t^0$ is concave in $\tau_t$, on all of $[0,1)$ when $m \leq 1$, that is $\xi(1-2\alpha) \leq 1$. Every term enters $G_t$ with a positive weight, so $G_t$ is continuous and strictly decreasing on $[0,1)$, finite at zero and unbounded below as $\tau_t \to 1$.

At fixed shares $x$ the second term of \eqref{eq:LOG:E2i}, $ap_i/(x_i+a\tau_tp_i)$, is strictly decreasing in $\tau_t$ wherever the retiree's consumption is positive, so $\mathcal{F}_t(\cdot\,;x)$ is strictly decreasing and the political objective strictly concave in $\tau_t$ there. A tax $\tau^{\ast}$ therefore maximizes the objective at the shares $x(\tau^{\ast})$ if and only if $\mathcal{F}_t(\tau^{\ast};x(\tau^{\ast})) = G_t(\tau^{\ast}) = 0$, or $\tau^{\ast} = 0$ and $G_t(0) \leq 0$. Exactly one tax does so, the zero of $G_t$ if $G_t(0) > 0$ and $\tau_t = 0$ otherwise. Every other candidate of the test is a corner at which $G_t \neq 0$, so the objective at its own shares rises as the tax moves away from it, and it fails.

At $T$ the young have no second period and their terms are \eqref{eq:LOG:E10}--\eqref{eq:LOG:E1i} with $\beta = 0$ (appendix \ref{app:T}), so $G_T$ involves no state and neither does the tax at $T$. Given a continuation that does not respond to the state, the young's terms at $t$ take the form \eqref{eq:LOG:E10}--\eqref{eq:LOG:E1i}, and $G_t$ again involves no state, so by induction this holds at every date. The continuation enters $G_t$ only through the level of $\Theta_{h,t}$ in $b_t^0$, by \eqref{eq:h}. Backward induction from $T$ therefore gives a unique equilibrium for every horizon, and without informal households $G_t$ involves only $\tau_t$, $\theta_t$, $\nu_t$ and the parameters. $\square$

The argument carries over to the tax condition with the cost of section \ref{sec:esc} at the inherited design (appendix \ref{app:esc:equilibrium}, where $\gamma_0 = 0$). With $u(\tau_t) \equiv \tau_tf(\theta_t,\tau_t)$, by \eqref{eq:esc:E2} and \eqref{eq:esc:auxiliary:si} the benefit term at consistent shares becomes $ap_iu'(\tau_t)/(y_i+aq_iu(\tau_t))$, with $u' = f(1+\ln f)$. Since $\ln f$ is linear in $\tau_t$, $u'' \leq 0$ where $u' \geq 0$, so the term is strictly decreasing there and negative elsewhere, and $G_t$ is strictly decreasing where it can vanish and negative beyond. At fixed shares the condition likewise changes sign at most once, from positive to negative, so the tax function $\mathcal{T}_t(\theta_t)$ of appendix \ref{app:esc:equilibrium} is single-valued. The proposition does not extend to informal savers or to CRRA preferences. With either, the condition carries the response of the continuation policy to the state, which has no fixed sign, and under CRRA preferences every term carries a marginal-utility weight that moves with the tax. For these cases the solver records at every state and date how many candidates pass the test (appendix \ref{app:T}).

\smalltitle{Comparative statics.} By the proof of proposition \ref{prop:PEE:unique}, $G_t$, and $\mathcal{F}_t$ at fixed shares, have strictly negative derivatives in $\tau_t$, since that of \eqref{eq:LOG:E1i} is strictly negative and those of the other terms are not positive. At an interior equilibrium a parameter therefore moves the tax in the direction in which it moves $G_t$, or $\mathcal{F}_t$ when the shares are held at their inherited values, by the implicit function theorem, and at the corner $\tau_t = 0$ a small change has no effect. The comparative statics are those of the tax at $t$ given the continuation policy, which enters only through $\sigma_t^0$. Without a universal pension it does not enter, and they hold at every date of the equilibrium path.

\smalltitle{$\omega$ and $\nu_t$.} The retirees' terms sum to a positive number at an interior equilibrium, since the young's are negative, so taxes rise with $\omega$. The ratio $\nu_t$ weights the young's terms and also enters the universal pension. The pension $b_t^0$ is proportional to $\nu_t$, more contributors per informal retiree, while the endowment is not, and both scale with the same power of $s_{t-1}/\nu_t$ (appendix \ref{app:EE:CRRA}). Hence $\partial\sigma_t^0/\partial\nu_t = \sigma_t^0(1-\sigma_t^0)/\nu_t$ and $\partial\mathcal{E}_{2,t}^0/\partial\nu_t = \mathcal{E}_{2,t}^0(1-\sigma_t^0)/\nu_t$, while $\sigma_t^i$ does not move, since formal pensions and capital income both scale with $\nu_t$. Using $G_t = 0$ to substitute for the young's terms,
\begin{align}
    \frac{\partial G_t}{\partial\nu_t} &= \frac{\omega}{\nu_t}\gamma_0\mu_0\mathcal{E}_{2,t}^0\left(1-\sigma_t^0\right)+\gamma_0\mu_0\mathcal{E}_{1,t}^0+\sum_{i>0}\gamma_i\mu_i\mathcal{E}_{1,t}^i \notag\\
    &= -\frac{\omega}{\nu_t}\left[\gamma_0\mu_0\mathcal{E}_{2,t}^0\sigma_t^0+\sum_{i>0}\gamma_i\mu_i\mathcal{E}_{2,t}^i\right]. \label{eq:LOG:nuEffect}
\end{align}
Since $0 \leq \sigma_t^0 \leq 1$, the bracket is at least the smaller of the formal retirees' term $\sum_{i>0}\gamma_i\mu_i\mathcal{E}_{2,t}^i$ and the retirees' total, which is positive. Taxes therefore fall with $\nu_t$ whenever the aggregate marginal support of formal retirees is positive at the equilibrium, which by \eqref{eq:LOG:E2sigma} is
\begin{align}\label{eq:LOG:nuCondition}
    \sum_{i>0}\gamma_i\mu_i\mathcal{E}_{2,t}^i > 0 \quad\Longleftrightarrow\quad \frac{\sum_{i>0}\gamma_i\mu_i\sigma_t^i}{\sum_{i>0}\gamma_i\mu_i} > 1-\phi_t:
\end{align}
the average share of the pension in the consumption of formal retirees, weighted by their propensities to vote, exceeds the elasticity of the tax base. Without a universal pension the formal retirees' term is the retirees' total and the condition always holds. With one it fails only if the tax is sustained by informal retirees against formal ones.


\smalltitle{Income inequality.} For any two types with $y_H > y_L$, raise $\eta_H$ and lower $\eta_L$ such that
\begin{align*}
    \frac{\partial \eta_L}{\partial \eta_H} = - \frac{\gamma_H \left(\frac{\eta_H}{X_H}\right)^{\xi}}{\gamma_L \left(\frac{\eta_L}{X_L}\right)^{\xi}},
\end{align*}
which keeps $\Gamma_h$, and so aggregate labor supply and savings, constant for given policies. Since $\partial y_m/\partial\eta_m = (1+\xi)(\eta_m/X_m)^{\xi}/\Gamma_h$, the change leaves every other $y_j$ unchanged and satisfies $\gamma_H\,dy_H+\gamma_L\,dy_L = 0$: it transfers relative income from a poorer to a richer type at a constant mean. It reaches $G_t$ only through the pension shares of formal retirees, since the young's terms and $\phi_t$ do not involve the $y_i$ and $\sigma_t^0$ involves them only through $\Gamma_h$. At consistent shares, by \eqref{eq:LOG:consistent}, a formal retiree's pension share is a function of its relative income alone,
\begin{align}\label{eq:LOG:sigmaY}
    \sigma(y) = \frac{a\tau_t\left[\theta_ty+1-\theta_t\right]}{D(y)}, \qquad \sigma'(y) = -\frac{\kappa_t(1-\theta_t)}{D(y)^2}, \qquad \kappa_t \equiv a\tau_t\left(1+\frac{a\tau_t}{1+\beta}\right),
\end{align}
where $D(y) = y+a\tau_tq(y)$, with $q(y)$ defined as $q_i$ in \eqref{eq:LOG:consistent}, is proportional to the consumption of a retiree with relative income $y$ and rises with it. The share falls with relative income, and by less at higher incomes. Using $\gamma_L\,dy_L = -\gamma_H\,dy_H$, the transfer changes the retirees' terms by
\begin{align*}
    \frac{\omega}{\tau_t}\gamma_H\,dy_H\left[\mu_H\sigma'(y_H)-\mu_L\sigma'(y_L)\right] = \frac{\omega}{\tau_t}\gamma_H\,dy_H\,\kappa_t\left(1-\theta_t\right)\left[\frac{\mu_L}{D_L^2}-\frac{\mu_H}{D_H^2}\right],
\end{align*}
which is positive with equal propensities to vote, since $D_H > D_L$, and in general whenever $\mu_H/\mu_L < (c_{2,t}^H/c_{2,t}^L)^2$. Every spread of relative incomes built from such transfers therefore raises the tax. The result rests on the response of savings: with the shares held at their inherited values the transfer would move $\sigma_t^i$ only through the earnings-related part of the pension, $\partial\sigma_t^i/\partial y_i = a\tau_t\theta_tx_i/D_i^2$, and would leave the tax unchanged at $\theta_t = 0$.

\smalltitle{The design.} The design $\theta_t$ pays current retirees, who anticipated it when they saved. At consistent shares $\partial D_i/\partial\theta_t = \frac{\beta}{1+\beta}a\tau_t(y_i-1)$ by \eqref{eq:LOG:consistent}: consumption moves by the fraction $\beta/(1+\beta)$ of the change in the pension, savings having absorbed the rest. Since $D_i-\frac{\beta}{1+\beta}a\tau_tp_i = y_i\left(1+\frac{a\tau_t}{1+\beta}\right)$, also by \eqref{eq:LOG:consistent},
\begin{align}
    \frac{\partial\sigma_t^i}{\partial\theta_t} &= \frac{a\tau_t(y_i-1)}{D_i^2}\left(D_i-\frac{\beta}{1+\beta}a\tau_tp_i\right) = \kappa_t\frac{y_i}{D_i^2}\left(y_i-1\right), \notag\\
    \frac{\partial G_t}{\partial\theta_t} &= \frac{\omega\kappa_t}{\tau_t}\sum_{i>0}\gamma_i\frac{\mu_iy_i}{D_i^2}\left(y_i-1\right). \label{eq:LOG:thetaEffect}
\end{align}
Since $\sum_{i>0}\gamma_i(y_i-1) = 0$, the sum equals $\frac{1}{2}\sum_{i,j}\gamma_i\gamma_j\big(\frac{\mu_iy_i}{D_i^2}-\frac{\mu_jy_j}{D_j^2}\big)(y_i-y_j)$, as in the proof of proposition \ref{prop:esc:corner}, and is negative when $\mu_iy_i/D_i^2$ falls with relative income and incomes are not all equal. Taxes therefore fall with $\theta_t$ whenever $\mu_iy_i/(c_{2,t}^i)^2$ is decreasing in relative income across formal types. With equal propensities, $D$ is affine in $y$ by \eqref{eq:LOG:consistent}, and
\begin{align*}
    \frac{d\ln\left(y/D(y)^2\right)}{dy} = \frac{1}{y}-\frac{2D'(y)}{D(y)} = \frac{1}{yD(y)}\left(\frac{2\beta}{1+\beta}a\tau_t(1-\theta_t)-D(y)\right)
\end{align*}
is negative wherever $D(y) > \frac{2\beta}{1+\beta}a\tau_t(1-\theta_t)$. Since $D$ rises with $y$, the condition holds across formal types if every formal retiree consumes more than $2\beta/(1+\beta)$ times the flat component of its pension, $(1-\theta_t)h_{t-1}\overline{b}_t = R_ts_{t-1}a\tau_t(1-\theta_t)$. This is the case whenever no formal type borrows against its pension: with $x_i \geq 0$, $D_i \geq a\tau_tp_i \geq a\tau_t(1-\theta_t)$, which exceeds $\frac{2\beta}{1+\beta}a\tau_t(1-\theta_t)$ for $\beta < 1$ and $\theta_t < 1$, and the bound is zero at $\theta_t = 1$. With the shares held at their inherited values, as for a design that retirees did not anticipate, $\partial\sigma_t^i/\partial\theta_t = a\tau_t(y_i-1)x_i/D_i^2$, and the weight $x_i/(c_{2,t}^i)^2$ need not fall with income, since $x_i$ rises with income faster than $y_i$ when $\theta_t < 1$; the sign is then open. A permanent change in the design also moves the continuation, which enters $G_t$ only through $\sigma_t^0$ and so not at all without a universal pension.



\smalltitle{The universal pension.} The effect of $\epsilon$ works through retirees alone, since the young's terms do not depend on it. We sign it for a change that current retirees did not anticipate, with their savings shares at their inherited values, as at the reform date of section \ref{sec:argentina}, and with $\Theta_{h,t}$, which moves with the universal pension the young anticipate for themselves, held fixed. Benefits are shares of revenue $\nu_tw_th_t\tau_t$, $\epsilon/(1+\gamma_0\epsilon)$ per informal retiree and $p_i/(1+\gamma_0\epsilon)$ per formal retiree of type $i$, so a higher $\epsilon$ raises $b_t^0$ by $1/(\epsilon(1+\gamma_0\epsilon))$ in proportion and lowers every $b_t^i$ by $\gamma_0/(1+\gamma_0\epsilon)$ in proportion, a pure transfer since there are $\gamma_0$ informal retirees per formal one. A pension share responds to a proportional change in the benefit by $\partial\sigma_t^j/\partial\ln b_t^j = \sigma_t^j(1-\sigma_t^j)$, since the rest of consumption does not move, and by \eqref{eq:LOG:E2sigma} $\partial\mathcal{E}_{2,t}^0/\partial\sigma_t^0 = \phi_t/\tau_t$ and $\partial\mathcal{E}_{2,t}^i/\partial\sigma_t^i = 1/\tau_t$. Hence
\begin{align*}
    \dfrac{\partial \mathcal{F}_t}{\partial \epsilon} =\frac{\omega\,\gamma_0}{\tau_t(1+\gamma_0\epsilon)}\left[\frac{\phi_t}{\epsilon}\,\mu_0\sigma_t^0\big(1-\sigma_t^0\big)-\sum_{i>0}\gamma_i\mu_i\sigma_t^i\big(1-\sigma_t^i\big)\right],
\end{align*}
and taxes rise with $\epsilon$ if and only if the bracket is positive,
\begin{align}\label{eq:LOG:epsCondition}
    \frac{\phi_t}{\epsilon}\,\mu_0\sigma_t^0\big(1-\sigma_t^0\big) > \sum_{i>0}\gamma_i\mu_i\sigma_t^i\big(1-\sigma_t^i\big).
\end{align}
The share $\gamma_0$ multiplies the whole expression, because the loss of formal retirees is the gain of informal retirees times their number, and enters the bracket only through the consumption levels, so it does not determine the sign. The factor $\sigma(1-\sigma)$ is zero for a retiree without a pension and for one who lives on the pension alone, whose consumption and marginal benefit a higher $\epsilon$ raises in the same proportion, and is largest at one half. The factor $1/\epsilon$ is the asymmetry of the transfer: a unit of $\epsilon$ is a large proportional change of a small universal pension and a small one of the formal pensions. With equal pension shares and equal propensities to vote the condition is $\epsilon < \phi_t$, and since informal retirees then consume $\epsilon$ times the average consumption of formal retirees, it requires them to be poorer by more than $1-\phi_t$, the elasticity of the tax base. The condition therefore holds when informal retirees are poorer than formal ones by more than this elasticity and the pension is a comparable share of their consumption, since a comparable share of a smaller consumption is a smaller pension, and fails when they live on the pension alone or are much richer than formal retirees. A permanent reform also lowers the hours of the young (part i of proposition \ref{prop:LOG:EE}), and with them the universal pension relative to the informal endowment, and a change that retirees anticipated would also move their savings shares; \eqref{eq:LOG:epsCondition} leaves out both.

At the Argentine calibration of section \ref{sec:argentina}, in 2010 before the reform, $\epsilon = 0.21$, $\phi_t = 0.976$, and the pension is $12.2\%$ of the consumption of informal retirees and between $19.5\%$ and $15.5\%$ of that of formal retirees, falling with income. The left-hand side of \eqref{eq:LOG:epsCondition} is then $0.498$, three and a half times the right-hand side, $0.143$. After the reform, at $\epsilon^U = 0.546$ and the post-reform tax of $13.8\%$, at which $\phi_t = 0.971$, the pension is $26.9\%$ of the consumption of informal retirees and the two sides are $0.350$ and $0.145$. The calibration has informal savers, whose savings income falls with hours as that of formal retirees does, so their term carries a factor of one in place of $\phi_t$, which raises the left-hand side before the reform to $0.511$.

\smalltitle{The representative-agent tax.} With one type of formal household and no informal households, $J = 1$ and $\gamma_0 = 0$, relative income, the savings share and the relative pension are one, $y_1 = x_1 = p_1 = 1$, so the design drops out of \eqref{eq:LOG:E2i}: both components of the benefit pay the same household. The propensity to vote multiplies both generations' terms and cancels, and with $a = (1-\alpha)/\alpha$ the second term of \eqref{eq:LOG:E2i} is $a/(1+a\tau_t)$, so the first order condition is
\begin{align*}
    \omega\left[\frac{1-\alpha}{\alpha+(1-\alpha)\tau_t}-\frac{(1-\alpha)\xi}{(1+\alpha\xi)(1-\tau_t)}\right] = \frac{\nu_t}{1-\tau_t}\frac{1+\xi}{1+\alpha\xi}\left(1+\beta\frac{\alpha(1+\xi)}{1+\alpha\xi}\right).
\end{align*}
Multiplying by $(1-\tau_t)(1+\alpha\xi)/(1-\alpha)$ and writing $\Psi \equiv (1+\xi)\big(1+\frac{\alpha\beta(1+\xi)}{1+\alpha\xi}\big)$ gives $(1+\alpha\xi)(1-\tau_t)/(\alpha+(1-\alpha)\tau_t) = \xi+\nu_t\Psi/(\omega(1-\alpha))$, which is linear in $\tau_t$ once cross-multiplied, and the unique solution is
\begin{align}\label{eq:LOG:repAgentTax}
    \tau_t = \max\left\{0,\;\frac{\omega(1-\alpha)-\alpha\nu_t\Psi}{\omega(1-\alpha)(1+\xi)+(1-\alpha)\nu_t\Psi}\right\}.
\end{align}
Where it is positive, the tax rises with $\omega$ and falls with $\nu_t$, $\alpha$, $\beta$ and $\xi$. With inelastic labor, $\xi = 0$, $\Psi = 1+\alpha\beta$ and \eqref{eq:LOG:repAgentTax} is the equilibrium tax of \textcite{mkvss08}, their ratio of young to old households being $\nu_t$ and their per capita political weight of retirees $\omega$. The Frisch elasticity lowers the tax because GHH preferences carry no income effect: the tax lowers hours, which erodes the base that pays retirees and deepens the loss of the young, whereas under the separable logarithmic preferences of their section 3 a social security tax that is the only tax leaves hours unchanged. The same expression is the tax at the Bismarckian corner for any income distribution and any propensities to vote, as long as there are no informal households. At $\theta_t = 1$ benefits and, by part iv of proposition \ref{prop:LOG:EE}, savings are proportional to relative income, so every type has $\sigma_t^i = a\tau_t/(1+a\tau_t)$ and the same marginal effects, and the propensities cancel. At the U.S.\ calibration of section \ref{sec:oecd} it gives a tax of $8.3\%$ in 2020, against the calibrated $10.5\%$.



\subsection{With log-GHH preferences and informal savings}\label{app:PEE:LOG_s0}


The introduction of $s_{t-1,0}/s_{t-1}$ as an endogenous state variable complicates the model because this state variable not only depends on past choices (through the effect of $\tau_{t-1}$ on $\Theta_{s,t-1}$), but is also forward looking ($s_{t-1,0}/s_{t-1}$ depends explicitly on $\tau_t$). To deal with this, let $dx_t/d\tau_t$ denote the effect in politico-economic equilibrium of changing the tax rate $\tau_t$ on the variable $x_t$ accounting for the fact that a change in $\tau_t$ also changes the state $s_{t,0}/s_t$ which through the continuation policy affects the choice of $\tau_{t+1}$ as well. Specifically, for the state variable $s_{t,0}/s_t$ we have:
\begin{align*}
    \frac{d(s_{t,0}/s_t)}{d\tau_t} = \frac{\partial (s_{t,0}/s_t)}{\partial \tau_t} + \frac{\partial (s_{t,0}/s_t)}{\partial \tau_{t+1}} \frac{d\tau_t^{+1}}{d(s_{t,0}/s_t)}\frac{d(s_{t,0}/s_t)}{d\tau_t},
\end{align*}
where the partial derivatives of $s_{t,0}/s_t$ capture how the state reacts to policy in economic equilibrium and $\tau_t^{+1}(s_{t,0}/s_t)$ is the continuation policy that details how policy in $t+1$ adjusts when the endogenous state changes.
When considering a marginal change in $\tau_t$, the policymaker takes into account how this propagates via economic equilibrium and continuation policy functions into future periods in order to summarize the total impact on current relevant variables. The technical documentation derives a recursive structure for all the relevant derivatives required to evaluate the marginal effect of $\tau_t$ on the indirect utility of households \parencite[section \textit{Informal Savings Model}]{BergGE26doc}. 


With the endogenous state $s_{t-1,0}/s_{t-1}$, the politico-economic equilibrium is now characterized by the first order condition:
\begin{align*}
    \omega\Bigg[\sum_{i\geq0}\gamma_i\mu_i \overbrace{\dfrac{d \ln\left(c_{2,t}^i\right)}{d \tau_t}}^{= \mathcal{E}_{2,t}^i}\Bigg]+
  \nu_t\Bigg\lbrace \sum_{i\geq 0}\gamma_i\mu_i\underbrace{\left(\dfrac{d \ln\left(\tilde{c}_{1,t}^i\right)}{d \tau_t}+\beta\dfrac{d \ln\left(c_{2,t+1}^i\right)}{d \tau_t}\right)}_{= \mathcal{E}_{1,t}^i}\Bigg\rbrace \leq 0,
\end{align*}
Using equilibrium consumption functions from appendix \ref{app:EE:CRRA}, we have the derivatives of indirect utility for retired households:
\begin{align}
    \mathcal{E}_{2,t}^0 & = (1-\alpha)\dfrac{d\ln\left(h_t\right)}{d\tau_t}+\frac{\frac{1-\alpha}{r_0\alpha}\frac{\epsilon}{1+\gamma_0\epsilon}}{\frac{s_{t-1,0}}{s_{t-1}}+\frac{1-\alpha}{r_0\alpha}\frac{\epsilon\tau_t}{1+\gamma_0\epsilon}} \\ 
    \mathcal{E}_{2,t}^i &=(1-\alpha)\frac{d\ln\left(h_t\right)}{d\tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{1}{1+\gamma_0\epsilon}\left(1+\theta_t\left[\frac{\eta_{i}^{1+\xi}}{X_{i}^{\xi}}\frac{1}{\Gamma_{h}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{\tau_t}{1+\gamma_0\epsilon}\left(1+\theta_t\left[\frac{\eta_{i}^{1+\xi}}{X_{i}^{\xi}}\frac{1}{\Gamma_{h}}-1\right]\right)}.
\end{align}
We note that the marginal effect for formal retirees is the same as when continuation policies were independent of state variables. The only difference is that the change in the aggregate labor supply, $\frac{d\ln\left(h_t\right)}{d\tau_t}$, now includes strategic effects through future policies. 

For the formal, working households, we can use the Euler equation to write the indirect utility as $\ln\left(\tilde{c}_{1,t}^i\right)\left(1+\beta\right)+\beta\ln\left(\beta R_{t+1}\right)$. The marginal effect on indirect utility then consists of the two terms:
\begin{align*}
    \mathcal{E}_{1,t}^i =& (1+\beta)\dfrac{d\ln\left(\tilde{c}_{1,t}^i\right)}{d\tau_t} + \beta \dfrac{d \ln\left(R_{t+1}\right)}{d\tau_t} \\
    \dfrac{d\ln\left(\tilde{c}_{1,t}^i\right)}{d\tau_t} =& \frac{1+\xi}{\xi}\dfrac{d\ln\left(h_t\right)}{d\tau_t}+\frac{\Gamma_{s,t}\frac{1-\alpha}{\alpha}\frac{1-\theta_{t+1}}{1+\gamma_0\epsilon}\left(\frac{d\tau_{t+1}}{d\tau_t}+\tau_{t+1}\frac{d\ln\left(\Gamma_{s,t}\right)}{d\tau_t}\right)}{\frac{\eta_{i}^{1+\xi}}{X_{i}^{\xi}}\frac{1}{1+\xi}+\Gamma_{s,t}\frac{1-\alpha}{\alpha}\frac{\tau_{t+1}(1-\theta_{t+1})}{1+\gamma_0\epsilon}} \\ 
    \dfrac{d \ln\left(R_{t+1}\right)}{d\tau_t} =& (1-\alpha)\left(\frac{d\ln\left(h_{t+1}\right)}{d\ln\left(s_t\right)}-1\right)\frac{d\ln\left(s_t\right)}{d\tau_t}.
\end{align*}
where we use the shorthand $d\tau_{t+1}/d\tau_t = \frac{d \tau_t^{+1}}{d (s_{t,0}/s_t)}\frac{d (s_{t,0}/s_t)}{d\tau_t}$ to define the total effect of a change in the tax rate on future taxes. The technical documentation gives the computation of these derivatives in the numerical application \parencite[section \textit{Informal Savings Model}]{BergGE26doc}.

For the informal, working households, a similar Euler condition applies such that we can write indirect utility as $\ln\left(\tilde{c}_{1,t}^0\right)\left(1+\beta\right)+\beta \ln\left(\beta R_{t+1}^0\right)$.
The marginal effect on indirect utility is then characterized by
\begin{align*}
    \mathcal{E}_{1,t}^0 =& (1+\beta)\dfrac{d\ln\left(\tilde{c}_{1,t}^0\right)}{d\tau_t} + \beta\dfrac{d \ln\left(R_{t+1}^0\right)}{d\tau_t} \\
    \frac{d\ln\left(\tilde{c}_{1,t}^0\right)}{d\tau_t} =& \frac{\frac{1-\alpha}{r_0\alpha}\frac{\epsilon}{1+\gamma_0\epsilon}\Theta_{s,t}\left(\frac{d\tau_{t+1}}{d\tau_t}+\tau_{t+1}\frac{d\ln\left(s_t\right)}{d\tau_t}\right)}{\frac{\eta_{0}^{1+\xi}}{X_{0}^{\xi}}\frac{1}{1+\xi}\left(\frac{1-\alpha}{\Gamma_{h}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\frac{1-\alpha}{r_0\alpha}\frac{\epsilon\tau_{t+1}}{1+\gamma_0\epsilon}\Theta_{s,t}} \\
    \frac{d\ln\left(R_{t+1}^0\right)}{d\tau_t} =& (1-\alpha)\left(\frac{d\ln\left(h_{t+1}\right)}{d\ln\left(s_t\right)}-1\right)\frac{d\ln\left(s_t\right)}{d\tau_t}
\end{align*}



\subsection{With CRRA-GHH preferences and informal savings}\label{app:PEE:CRRA}

With CRRA preferences it is no longer the case that future policy is independent of the capital stock. But conditional on this state variable, with GHH preferences it is still the case that the distribution of savings for formal workers, $\{s_{t-1,i}/s_{t-1}\}_{i>0}$, remains a function of $s_{t-1}$ and $\tau_t$ and is thus not an independent state variable. Thus, the policy function can be written as a function of two endogenous states: $s_{t-1}$ and $s_{t-1,0}/s_{t-1}$. 

Compared to the log-GHH case, we adjust the definition of the politico-economic equilibrium derivative $dx_t/d\tau_t$ to include the effect through both states such that 
\begin{align*}
    \dfrac{d x_t}{d \tau_t} = \dfrac{\partial x_t}{\partial \tau_t}+\frac{\partial x_t}{\partial \tau_{t+1}}\left(\frac{\partial \tau_t^{+1}}{\partial s_t}\frac{d s_t}{d \tau_t}+\frac{\partial \tau_t^{+1}}{\partial (s_{t,0}/s_t)}\frac{ d (s_{t,0}/s_t)}{d\tau_t}\right).
\end{align*}

The first order condition identifying the politico-economic choice $\tau_t$ is now given by
\begin{align*}
    &\omega \sum_{j\geq 0} \gamma_j\mu_j\underbrace{ \left(c_{2,t}^j\right)^{\frac{\rho-1}{\rho}}\frac{d\ln\left(c_{2,t}^j\right)}{d\tau_t}}_{\equiv \mathcal{E}_{2,t}^j} \\+&\nu_t\sum_{j\geq0}\gamma_j\mu_j\underbrace{\left(\left(\tilde{c}_{1,t}^j\right)^{\frac{\rho-1}{\rho}}\frac{d\ln\left(\tilde{c}_{1,t}^j\right)}{d\tau_t}+\beta \left(c_{2,t+1}^j\right)^{\frac{\rho-1}{\rho}}\frac{d\ln\left(c_{2,t+1}^j\right)}{d\tau_t}\right)}_{\equiv \mathcal{E}_{1,t}^j}
    \leq 0,
\end{align*}

The log-derivatives $d\ln\left(c_{2,t}^j\right)/d\tau_t$ follow from the formulas for the log-GHH case and consumption levels $c_{2,t}^j$ depend on the state $s_{t-1}$ and are given in appendix \ref{app:EE:CRRA}.

For formal working households, we use the Euler equation to write indirect utility as
\begin{align*}
    v_{1,t}^i =& (1+B_{t+1})\frac{\Big(\tilde{c}_{1,t}^i\Big)^{1-1/\rho}}{1-1/\rho} = \frac{\left(\hat{c}_{1,t}^i\right)^{1-1/\rho}}{1-1/\rho} \\
    \hat{c}_{1,t}^i =& \left(\frac{h_t}{\Gamma_h}\right)^{\frac{1+\xi}{\xi}}(1+B_{t+1})^{\frac{1}{\rho-1}}\mathcal{C}_{1,t}^i \\ 
    \mathcal{C}_{1,t}^i =& \frac{\eta_i^{1+\xi}}{X_i^{\xi}}\frac{1}{1+\xi}+\Gamma_{s,t}\frac{1-\alpha}{\alpha}\frac{\tau_{t+1}(1-\theta_{t+1})}{1+\gamma_0\epsilon}
\end{align*}
The marginal effect of taxes can then be summed up by
\begin{align}
    \frac{dv_{1,t}^i}{d\tau_t}=& \left(\hat{c}_{1,t}^i\right)^{1-1/\rho} \frac{d\ln\left(\hat{c}_{1,t}^i\right)}{d\tau_t} \\
    \frac{d\ln\left(\hat{c}_{1,t}^i\right)}{d\tau_t} =& \frac{1+\xi}{\xi}\dfrac{d\ln\left(h_t\right)}{d\tau_t}+\frac{B_{t+1}}{1+B_{t+1}}(1-\alpha)\left(\frac{d\ln\left(h_{t+1}\right)}{d\ln\left(s_t\right)}-1\right)\frac{d\ln\left(s_t\right)}{d\tau_t} \notag\\
    &+\frac{\Gamma_{s,t}\frac{1-\alpha}{\alpha}\frac{1-\theta_{t+1}}{1+\gamma_0\epsilon}\left(\frac{d\tau_{t+1}}{d\tau_t}+\tau_{t+1}\frac{d\ln\left(\Gamma_{s,t}\right)}{d\tau_t}\right)}{\frac{\eta_i^{1+\xi}}{X_i^{\xi}}\frac{1}{1+\xi}+\Gamma_{s,t}\frac{1-\alpha}{\alpha}\frac{\tau_{t+1}(1-\theta_{t+1})}{1+\gamma_0\epsilon}} 
\end{align}


For the young informal households, we can employ the Euler equation in a similar way and write:
\begin{align*}
    v_{1,t}^0 =& \frac{\Big(\tilde{c}_{1,t}^0(1+B_{t+1}^0)^{\frac{\rho}{\rho-1}}\Big)^{1-1/\rho}}{1-1/\rho} = \frac{\left(\hat{c}_{1,t}^0\right)^{1-1/\rho}}{1-1/\rho} \\
    \hat{c}_{1,t}^0 =& \left(\frac{s_{t-1}}{\nu_t}\right)^{\sigma_s}(1+B_{t+1}^0)^{\frac{1}{\rho-1}}\mathcal{C}_{1,t}^0 \\ 
    \mathcal{C}_{1,t}^0 =& \frac{\eta_{0}^{1+\xi}}{X_{0}^{\xi}}\frac{1}{1+\xi}\left(\frac{1-\alpha}{\Gamma_{h}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\Theta_{s,t}\frac{1-\alpha}{r_0\alpha}\frac{\tau_{t+1}\epsilon}{1+\gamma_0\epsilon}
\end{align*}
We then have that:
\begin{align}
    \frac{dv_{1,t}^0}{d\tau_t}=& \left(\hat{c}_{1,t}^0\right)^{1-1/\rho} \frac{d\ln\left(\hat{c}_{1,t}^0\right)}{d\tau_t} \\
    \frac{d\ln\left(\hat{c}_{1,t}^0\right)}{d\tau_t} =& \frac{B_{t+1}^0}{1+B_{t+1}^0}(1-\alpha)\left(\frac{d\ln\left(h_{t+1}\right)}{d\ln\left(s_t\right)}-1\right)\frac{d\ln\left(s_t\right)}{d\tau_t} \notag\\
    &+\frac{\Theta_{s,t}\frac{1-\alpha}{r_0\alpha}\frac{\epsilon}{1+\gamma_0\epsilon}\left(\frac{d\tau_{t+1}}{d\tau_t}+\tau_{t+1}\frac{d\ln\left(\Theta_{s,t}\right)}{d\tau_t}\right)}{\frac{\eta_{0}^{1+\xi}}{X_{0}^{\xi}}\frac{1}{1+\xi}\left(\frac{1-\alpha}{\Gamma_{h}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\Theta_{s,t}\frac{1-\alpha}{r_0\alpha}\frac{\tau_{t+1}\epsilon}{1+\gamma_0\epsilon}} 
\end{align}
