\subsection{Finite Horizon Terminal state.}\label{\refsec:finitehorizon}

\smalltitle{Economic equilibrium.}
In the terminal period there are no savings, $s_t = s_{t,i} = 0$, and the economic equilibrium simplifies to
\begin{subequations}\label{\refeq:TerminalEE}
\begin{align}
    h_t &= \left(\Gamma_{h,t}\right)^{\frac{1}{1+\alpha\xi}}\left((1-\alpha)(1-\tau_t)\right)^{\frac{\xi}{1+\alpha\xi}}\left(\frac{s_{t-1}}{\nu_t}\right)^{\frac{\alpha\xi}{1+\alpha\xi}} \\
    h_t^i &= h_t\frac{(\eta_{t,i}/X_{t,i})^{\xi}}{\Gamma_{h,t}} \\
    c_{1,t}^i &= \dfrac{\eta_{t,i}^{1+\xi}}{X_{t,i}^{\xi}}\left(\frac{h_t}{\Gamma_{h,t}}\right)^{\frac{1+\xi}{\xi}} \\
    \tilde{c}_{1,t}^i &= \frac{c_{1,t}^i}{1+\xi} \\
    c_{1,t}^0 &= \frac{\eta_{t,0}^{1+\xi}}{X_{t,0}^{\xi}}\left(\frac{1-\alpha}{\Gamma_{h,t}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}\left(\frac{s_{t-1}}{\nu_t}\right)^{\frac{\alpha(1+\xi)}{1+\alpha\xi}} \\
    \tilde{c}_{1,t}^0 &= \dfrac{c_{1,t}^0}{1+\xi}.
\end{align}
\end{subequations}

\smalltitle{Politico-economic equilibrium with log preferences.}
The politico-economic equilibrium simplifies as well, since the terminal period has no continuation policy. In the log case the expressions of \eqref{\refeq:PEELOG} apply with $\beta_{t,j} = 0$, which simplifies the young's terms and leaves the old's unchanged:
\begin{subequations}\label{\refeq:terminalPEELOG}
\begin{align}
    \dfrac{d \upsilon_{1,t}^i}{d \tau_t} =& - \frac{1}{1-\tau_t}\frac{1+\xi}{1+\alpha\xi} \\
    \dfrac{d \upsilon_{1,t}^0}{d \tau_t} =& 0 \\
    \frac{d\upsilon_{2,t}^i}{d\tau_t} =& (1-\alpha)\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial \tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{p_{t-1}}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{p_{t-1}\tau_t}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)} \\
    \frac{d\upsilon_{2,t}^0}{d\tau_t} =& \frac{\dfrac{(1-\alpha)\nu_t\epsilon_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}\left(1+(1-\alpha)\tau_t\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial\tau_t}\right)}{\frac{\left(\chi_{t-1}\eta_{t-1,0}\right)^{1+\xi}}{(1+\xi)X_{t-1,0}^{\xi}}\left(\frac{1-\alpha}{\Gamma_{h,t}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\dfrac{(1-\alpha)\nu_t\epsilon_t\tau_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}},
\end{align}
with $\partial \ln\left(\Theta_{h,t}\right)/\partial \tau_t$ as for $t<T$.
\end{subequations}

\smalltitle{Politico-economic equilibrium with CRRA preferences.}
With $\rho\neq 1$ the savings level $s_{t-1}$ is a state, and the first order condition needs consumption levels to weigh the marginal effects. Otherwise the log formulation carries over, and the first order condition is written directly, without numerical grid searches:
\begin{subequations}\label{\refeq:terminalPEECRRA}
\begin{align}
    \dfrac{d \upsilon_{1,t}^i}{d \tau_t} =& - \frac{1}{1-\tau_t}\frac{1+\xi}{1+\alpha\xi}\left(\tilde{c}_{1,t}^i\right)^{1-1/\rho}  \\
    \dfrac{d \upsilon_{1,t}^0}{d \tau_t} =& 0 \\
    \frac{d\upsilon_{2,t}^i}{d\tau_t} =& \left(c_{2,t}^i\right)^{1-1/\rho}\left[(1-\alpha)\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial \tau_t}+\frac{\frac{1-\alpha}{\alpha}\frac{p_{t-1}}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)}{\frac{s_{t-1,i}}{s_{t-1}}+\frac{1-\alpha}{\alpha}\frac{p_{t-1}\tau_t}{\kappa_{t-1}}\left(1+\theta_t\left[\frac{\eta_{t-1,i}^{1+\xi}}{X_{t-1,i}^{\xi}}\frac{1}{\Gamma_{h,t-1}}-1\right]\right)}\right] \\
    \frac{d\upsilon_{2,t}^0}{d\tau_t} =& \left(\tilde{c}_{2,t}^0\right)^{1-1/\rho}\frac{\dfrac{(1-\alpha)\nu_t\epsilon_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}\left(1+(1-\alpha)\tau_t\frac{\partial \ln\left(\Theta_{h,t}\right)}{\partial\tau_t}\right)}{\frac{\left(\chi_{t-1}\eta_{t-1,0}\right)^{1+\xi}}{(1+\xi)X_{t-1,0}^{\xi}}\left(\frac{1-\alpha}{\Gamma_{h,t}^{\alpha}}\right)^{\frac{1+\xi}{1+\alpha\xi}}+\dfrac{(1-\alpha)\nu_t\epsilon_t\tau_t}{\kappa_{t-1}} \Theta_{h,t}^{1-\alpha}},
\end{align}
with $\partial \ln\left(\Theta_{h,t}\right)/\partial \tau_t$ as in the log case.
\end{subequations}
