\subsection{The politico-economic equilibrium with CRRA}\label{\refsec:PEE}
With CRRA preferences the past savings level $s_{t-1}$ becomes a state of the political problem: the marginal indirect utilities carry consumption \emph{levels}, and levels depend on the inherited stock. What is identified at each $t$ is therefore a policy \emph{function} $\tau_t(s_{t-1})$, built by backward recursion. The savings \emph{shares} $s_{t-1,i}/s_{t-1}$ remain closed form in $\tau_t$, so the aggregate level is the only state, and the selection of \cref{\refsec:RobustRoot} applies along the policy dimension unchanged. What is new is the grid over the state and a fixed point that pins the endogenous state down.

\smalltitle{Grids.} Let $\mathcal{S} = \left\lbrace s^{(1)}, ..., s^{(M_s)} \right\rbrace$ with $0<s^{(1)}<s^{(M_s)}$ denote a grid of states. The steady state $s^*(\tau)$ decreases in $\tau$, so the grid runs from a small positive $s^{(1)}$, which keeps the interest rate defined, to $1.25\times s^*(0)$. For $t<T$ we also need a grid $\mathcal{S}'$ of \emph{candidate} values of the endogenous state $s_t$: $\mathcal{S}$ indexes the states we solve \emph{at} and $\mathcal{S}'$ is searched \emph{over}, and the two are kept apart even when they hold the same values, the default, since the accuracy of the search depends on $\mathcal{S}'$ alone.

\smalltitle{The economic equilibrium at $t<T$ is a fixed point in $s_t$.}
Given continuation policies $\tau^{t+1}(\cdot)$ and $h^{t+1}(\cdot)$ from period $t+1$, both functions of the state $s_t$, the auxiliary equations \eqref{\refeq:auxiliary} at a candidate tax $\tau_t$ and a candidate state $s_t$ can be evaluated in one forward pass:
\begin{subequations}\label{\refeq:stateApprox}
\begin{align}
    \bm{B}_{t+1} &= \bm{B}_{t+1}\left(s_t,\, h^{t+1}(s_t)\right), \label{\refeq:stateApprox:B}\\
    \Gamma_{s,t} &= \Gamma_{s,t}\left(\bm{B}_{t+1},\, \tau^{t+1}(s_t),\, \theta_{t+1}\right), \label{\refeq:stateApprox:Gammas}\\
    \Theta_{h,t} &= \Theta_{h,t}\left(\tau_t,\, \tau^{t+1}(s_t),\, \theta_{t+1},\, \Gamma_{s,t}\right), \label{\refeq:stateApprox:Thetah}\\
    \Theta_{s,t} &= \Theta_{s,t}\left(\Theta_{h,t},\, \Gamma_{s,t}\right). \label{\refeq:stateApprox:Thetas}
\end{align}
\end{subequations}
The candidate $s_t$ is an equilibrium at $s_{t-1}$ when it reproduces itself through \eqref{\refeq:auxiliary:s}, that is, when it is a root of
\begin{align}\label{\refeq:stateResidual}
    \mathcal{R}_t\left(\tau_t, s_t;\, s_{t-1}\right) \equiv \Theta_{s,t}\left(\tau_t, s_t\right)\left(\frac{s_{t-1}}{\nu_t}\right)^{\frac{\alpha(1+\xi)}{1+\alpha\xi}}-s_t
\end{align}
in its second argument, which defines the state policy function $s_t(\tau_t, s_{t-1})$. This is a genuine root problem, not a recursion we can unroll, because $s_t$ enters the right-hand side through $\bm{B}_{t+1}$ and the continuation policies. Any crossing is a solution, so neither the equilibrium test nor the selection of \cref{\refsec:RobustRoot} applies; where several crossings exist, the lowest is taken and their number reported.

\smalltitle{The fixed point is two-dimensional, not three.}
None of $\bm{B}_{t+1}$, $\Gamma_{s,t}$, $\Theta_{h,t}$, $\Theta_{s,t}$ in \eqref{\refeq:stateApprox} depends on $s_{t-1}$, which enters \eqref{\refeq:stateResidual} only through the factor $(s_{t-1}/\nu_t)^{\alpha(1+\xi)/(1+\alpha\xi)}$. The costly part is therefore a function of $(\tau_t, s_t)$ alone, evaluated once on $\mathcal{T}\times\mathcal{S}'$, after which the residual for \emph{every} $s_{t-1}\in\mathcal{S}$ follows by scaling $\Theta_{s,t}$ and subtracting $s_t$: what looks like a three-dimensional problem costs a two-dimensional one plus a broadcast.

\smalltitle{Feasibility.}
The root of \eqref{\refeq:stateResidual} need not lie inside $\mathcal{S}'$: for a given $s_{t-1}$, a large $\tau_t$ depresses savings below the grid, and conversely at small $\tau_t$, and outside that window everything downstream of $s_t$ is undefined. We therefore carry a feasibility mask over $\mathcal{T}\times\mathcal{S}$, marking the pairs $(\tau_t, s_{t-1})$ with a root, and require at least two feasible $\tau_t$ at a state before selecting there. The feasible set is expected to be an interval in $\tau_t$, since $s_t$ decreases in $\tau_t$, but masking cells individually does not rely on it.

\smalltitle{Which derivatives may be taken on the grid.}
Once $s_t(\tau_t,s_{t-1})$ is known, all four marginal indirect utilities take the form (level)$^{1-1/\rho}\times$(log-derivative):
\begin{subequations}\label{\refeq:PEEcrraTerms}
\begin{align}
    \frac{d\upsilon_{1,t}^i}{d\tau_t} &= \left(\hat{c}_{1,t}^i\right)^{1-1/\rho}\frac{d\ln\left(\hat{c}_{1,t}^i\right)}{d\tau_t}, &
    \frac{d\upsilon_{1,t}^0}{d\tau_t} &= \beta_{t,0}\left(\tilde{c}_{2,t+1}^0\right)^{1-1/\rho}\frac{d\ln\left(\tilde{c}_{2,t+1}^0\right)}{d\tau_t}, \\
    \frac{d\upsilon_{2,t}^i}{d\tau_t} &= \left(c_{2,t}^i\right)^{1-1/\rho}\frac{d\ln\left(c_{2,t}^i\right)}{d\tau_t}, &
    \frac{d\upsilon_{2,t}^0}{d\tau_t} &= \left(\tilde{c}_{2,t}^0\right)^{1-1/\rho}\frac{d\ln\left(\tilde{c}_{2,t}^0\right)}{d\tau_t}.
\end{align}
\end{subequations}
The young informal term uses that $\tilde{c}_{1,t}^0$ depends on $s_{t-1}$ only \eqref{\refmodeleq:EE:c0}. For the young formal households, $B_{t+1}^i$ is itself a function of $\tau_t$ through $s_t$, so the factor $(1+B_{t+1}^i)$ in $\upsilon_{1,t}^i$ cannot be held fixed while differentiating, and we absorb it into the consumption level,
\begin{align}\label{\refeq:hatc1i}
    \hat{c}_{1,t}^i \equiv \left(1+B_{t+1}^i\right)^{\frac{1}{1-1/\rho}}\tilde{c}_{1,t}^i \qquad\Longrightarrow\qquad \upsilon_{1,t}^i = \frac{\left(\hat{c}_{1,t}^i\right)^{1-1/\rho}}{1-1/\rho},
\end{align}
so that one numerical derivative of $\ln(\hat{c}_{1,t}^i)$ picks up both channels. $\hat{c}_{1,t}^i$ is never formed as a level, whose exponent diverges as $\rho\rightarrow 1$: only $\left(1+B_{t+1}^i\right)\left(\tilde{c}_{1,t}^i\right)^{1-1/\rho}$ and $\ln\left(1+B_{t+1}^i\right)/(1-1/\rho)+\ln\tilde{c}_{1,t}^i$ are needed.

The log-derivatives of $h_t$, $\hat{c}_{1,t}^i$ and $\tilde{c}_{2,t+1}^0$ depend on $\tau_t$ through the continuation policies and have no closed form; they are taken along the $\tau_t$ dimension of the grid, as derivatives of smoothing splines. The other two are not:
\begin{itemize}
    \item $d\ln(c_{2,t}^i)/d\tau_t$ \textbf{must} come from its closed form \eqref{\refmodeleq:PEE}. A grid derivative would be wrong, not merely imprecise: $s_{t-1,i}/s_{t-1}$ varies along the grid while the policy maker takes it as predetermined, so the grid derivative would include a channel that does not belong in the first order condition.
    \item $d\ln(\tilde{c}_{2,t}^0)/d\tau_t$ \emph{may} be taken either way; we use its closed form for consistency.
\end{itemize}
Both closed forms are those of the log case, \eqref{\refmodeleq:PEELOG}, with $\partial\ln(\Theta_{h,t})/\partial\tau_t$ replaced by the numerical $d\ln(h_t)/d\tau_t$; the two coincide since $s_{t-1}$ is held fixed, so the log and CRRA cases share one formula.

\smalltitle{The terminal period, and the log limit.}
Under CRRA the terminal period needs no numerical differentiation: by the chain rule, each term of \eqref{\refmodeleq:terminalPEECRRA} is a consumption level raised to $1-1/\rho$ times the log-derivative of the log case. What is new is that $B_T^i$ is no longer the primitive $\beta_i$, so the terminal problem $z_T(\tau_T, s_{T-1})$ is state-dependent. The recursion at $t<T$ divides by $1-1/\rho$ and refuses $\rho = 1$, which \cref{\refsec:PEELOG} solves. The terminal period is defined at $\rho = 1$, where $\bm{B}_T$ collapses to $\bm{\beta}$ and every level factor to unity, and it then coincides \emph{exactly} with the terminal period of the log solver. Since the two solvers share almost no code, this identity, and the convergence of the CRRA solution to the log solution as $\rho\rightarrow 1$, is the sharpest available cross-check on both.

\begin{alg}[Backward grid search over the savings level]
\label{\refalg:CRRA:grid}
Fix $\bm{\theta},\bm{\epsilon}$ and the grids $\mathcal{T}$, $\mathcal{S}$, $\mathcal{S}'$. The recursion identifies the policy functions $\tau_t(s_{t-1})$ and the state functions $s_t(\tau_t, s_{t-1})$, backwards from $t=T$.

\smallskip\noindent\emph{Terminal period.} Every object in $z_T$ is closed form in $(\tau_T, s_{T-1})$. Evaluate $z_T$ on $\mathcal{T}\times\mathcal{S}$, apply the selection of \cref{\refsec:RobustRoot} along $\tau$ at each state to obtain $\tau_T(s_{T-1})$, and build the interpolants $\tau^T(\cdot)$ and $h^T(\cdot)$ that period $T-1$ calls.

\smallskip\noindent\emph{Recursion, $t<T$.} Given the interpolants $\tau^{t+1}(\cdot)$ and $h^{t+1}(\cdot)$:
\begin{enumerate}
    \item \emph{State approximation.} Evaluate the forward pass \eqref{\refeq:stateApprox} on $\mathcal{T}\times\mathcal{S}'$, form the residual \eqref{\refeq:stateResidual} for every $s_{t-1}\in\mathcal{S}$ by broadcasting, and locate its root along $s_t$, which gives $s_t(\tau_t, s_{t-1})$ on $\mathcal{T}\times\mathcal{S}$ and the feasibility mask. Only this step uses $\mathcal{S}'$ and the continuation interpolants.
    \item \emph{Current-period objects.} Evaluate on $\mathcal{T}\times\mathcal{S}$ hours, the savings shares $s_{t-1,i}/s_{t-1}$ at the previous period's parameters, and the consumption levels in \eqref{\refeq:PEEcrraTerms}.
    \item \emph{First order condition.} Take the three numerical log-derivatives along $\tau_t$ at each $s_{t-1}$ and the other two from their closed forms, and assemble $z_t$ on $\mathcal{T}\times\mathcal{S}$ with the political weights of \eqref{\refeq:LOG:fast}, infeasible cells left undefined.
    \item \emph{Selection.} Apply the selection of \cref{\refsec:RobustRoot} along $\tau_t$ at each $s_{t-1}$ over its feasible cells. Smooth the selected $\tau_t(s_{t-1})$ with a light smoothing spline, which removes small kinks inherited from the interpolated continuation policies that the next period's numerical derivatives would amplify. Re-solve the state fixed point at the selected taxes and build the interpolants $\tau^t(\cdot)$ and $h^t(\cdot)$ that period $t-1$ calls.
\end{enumerate}
\end{alg}
